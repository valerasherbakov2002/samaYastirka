    <script>

        // Инициализация Telegram Web App

        const tg = window.Telegram.WebApp;

        tg.expand(); // Открывает приложение на всю высоту


        // Функция переключения вкладок

        function switchTab(pageId, tabElement) {

            // Скрываем все страницы

            document.querySelectorAll('.page').forEach(page => page.classList.remove('active'));

            // Убираем активный класс у всех кнопок

            document.querySelectorAll('.tab').forEach(tab => tab.classList.remove('active'));

            

            // Показываем нужную страницу и делаем кнопку активной

            document.getElementById(pageId).classList.add('active');

            tabElement.classList.add('active');

        }


        // Получение данных пользователя из Telegram

        const user = tg.initDataUnsafe.user;

        

        if (user) {

            // Установка имени и аватара (как на скриншоте Снимок экрана ... 20.15.23.png)

            document.getElementById('user-name').innerText = user.first_name + (user.last_name ? ' ' + user.last_name : '');

            

            if (user.photo_url) {

                document.getElementById('user-avatar').src = user.photo_url;

            }


            // Вывод уникального ID под QR-кодом

            document.getElementById('user-id-text').innerText = 'ID: ' + user.id;


            // Генерация QR-кода на основе уникального Telegram ID пользователя

            new QRCode(document.getElementById("qrcode"), {

                text: user.id.toString(),

                width: 150,

                height: 150,

                colorDark : "#000000",

                colorLight : "#ffffff",

                correctLevel : QRCode.CorrectLevel.H

            });

        } else {

            // Если открыто не в Telegram (например, в обычном браузере)

            document.getElementById('user-name').innerText = 'Гость';

            new QRCode(document.getElementById("qrcode"), {

                text: "guest_123",

                width: 150,

                height: 150

            });

        }
